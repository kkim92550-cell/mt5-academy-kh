ផែនទីចាប់ផ្តើមសម្រាប់អ្នកថ្មីពីសូន្យ

ចាប់ផ្តើមពីទីនេះ៖ គួរស្គាល់អ្វីមុន រៀនអ្វីមុន និងប្រើវេបសាយយ៉ាងដូចម្តេច?

មិនចាំបាច់ភ័យថាស្មុគស្មាញទេ — គ្រាន់តែដើរតាម ៣ ផ្នែកខ្លីៗខាងក្រោមតាមលំដាប់ អ្នកនឹងយល់ច្បាស់ពីមូលដ្ឋាន MT5 និងចេះប្រើឧបករណ៍ទាំងអស់ក្នុងវេបសាយនេះភ្លាម។

ផ្នែកទី ១៖ ស្គាល់ពាក្យស្នូលទាំង ៥ ក្នុង ២ នាទី (មុនបើកក្រាហ្វ)

ចុចលើពាក្យនីមួយៗខាងក្រោមដើម្បីមើលការពន្យល់ខ្លីៗ និងឧទាហរណ៍គិតជាដុល្លារជាក់ស្តែង។

Pip / Point

ខ្នាតវាស់ចម្ងាយតម្លៃរត់

Pip ជាខ្នាតសម្រាប់វាស់ថាតម្លៃមាស ឬរូបិយប័ណ្ណឡើង ឬចុះបានប៉ុន្មាន។ 1 Pip ស្មើនឹង 10 Points លើ MT5។

ឧទាហរណ៍ជាក់ស្តែងលើមាស (XAU/USD)

មាសឡើងពី 2,650.00 ទៅ 2,652.00 = រត់បាន 20 Pips។ បើអ្នកដាក់ 0.01 Lot (1 Pip = $0.10) → ចំណេញ 20 × $0.10 = $2.00។

Auto-Doctor

ពេលត្រេដ Demo បាន ២០ ដៃ ហើយចង់ឱ្យប្រព័ន្ធរកកំហុស (ម៉ោងខាត / Revenge Trade)។

ច្បាប់ដែក ៤ ចំណុចដែលអ្នកថ្មីហាមបំពាន

1. ហាមដាក់លុយពិតមុនចេះគណនា Lot Size

បើមិនទាន់អាចគណនាថា SL 25 Pips លើដើមទុន $1,000 ត្រូវដាក់ប៉ុន្មាន Lot ទេ សូមហាត់លើ Demo និង Checklist Gate សិន។

2. ហាមបើក Demo $100,000

បង្កើតគណនី Demo MT5 ឱ្យមានទឹកប្រាក់ស្មើលុយពិតដែលអ្នកគ្រោងប្រើ (ឧ. $500 ឬ $1,000) ដើម្បីកុំឱ្យខូចទម្លាប់ដាក់ Lot។

3. ហាមបើកដៃដោយគ្មាន Stop Loss

រាល់ប្រតិបត្តិការត្រូវកំណត់ SL មុនចុចបញ្ជាជានិច្ច និងហាមអូស SL គេចពេលតម្លៃជិតប៉ះ។

4. ហាមចូលដៃសងសឹកពេលខាត (Revenge Trade)

ពេលខាតដល់ដែនកំណត់ប្រចាំថ្ងៃ 2% ត្រូវបិទក្រាហ្វឈប់សម្រាកភ្លាម រួចរង់ចាំថ្ងៃស្អែក។
